    }

  function chunkEntries(entries, maxN, maxChars) {
    var chunks = [], cur = [], cc = 0;
    for (var i = 0; i < entries.length; i++) {
      var len = entries[i].text.length;
      if (cur.length && (cur.length >= maxN || cc + len > maxChars)) { chunks.push(cur); cur = []; cc = 0; }
      cur.push(entries[i]); cc += len;
    }
    if (cur.length) chunks.push(cur);
    return chunks;
  }
  function parseTranslations(content, cues) {
    /* JSON/source 回显只能验证结构与来源绑定，不能证明译文语义对齐。 */
    var value;
    try { value = JSON.parse(content); } catch (_) { throw new Error('输出不是有效 JSON 对象'); }
    if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error('输出必须是 JSON 对象');
    var keys = Object.keys(value);
    if (keys.length !== 1 || keys[0] !== 'translations' || !Array.isArray(value.translations)) throw new Error('输出必须只含 translations 数组');
    var expected = {};
    cues.forEach(function (c) { expected[c.id] = c.source; });
    var result = {};
    value.translations.forEach(function (row) {
      if (!row || typeof row !== 'object' || Array.isArray(row)) throw new Error('translations 项必须是对象');
      var rk = Object.keys(row);
      if (rk.length !== 3 || !('id' in row) || !('source' in row) || !('text' in row)) throw new Error('每项必须只包含 id、source、text');
      if (typeof row.id !== 'number' || !Number.isSafeInteger(row.id)) throw new Error('编号必须为整数');
      if (!(row.id in expected)) throw new Error('输出含额外编号 ' + row.id);
      if (row.id in result) throw new Error('输出含重复编号 ' + row.id);
      /* 严格比对失败再走规范化比对（模型复写 source 常见无害变形：空白/全角空格/零宽字符——
       * canan FC 实测：编号 2037 因此类差异逐字比对误杀且二分到单行无法恢复） */
      if (row.source !== expected[row.id] && normSource(row.source) !== normSource(expected[row.id])) throw new Error('编号 ' + row.id + ' 的 source 与原稿不符');
      if (typeof row.text !== 'string') throw new Error('编号 ' + row.id + ' 的译文不是字符串');
      var text = row.text.replace(/\r\n?/g, '\n').replace(/\[SPEAKER_\d+\]:?\s*/g, '').trim();
      if (!text || /\n[ \t]*\n/.test(text) || text.indexOf('-->') >= 0 || text.indexOf('\x00') >= 0) throw new Error('编号 ' + row.id + ' 的译文为空或包含破坏 SRT 的格式');
      result[row.id] = text;
    });
    cues.forEach(function (c) { if (!(c.id in result)) throw new Error('译文缺少编号 ' + c.id); });
    return result;
  }
  /* 深度兜底恢复用：source 字段模型可能反复规范化（如全角化、空白变形、零宽字符注入），

  function parseTranslationsLenient(content, cues) {

      if (row.id in result) throw new Error('输出含重复编号 ' + row.id);

      if (typeof row.text !== 'string') throw new Error('编号 ' + row.id + ' 的译文不是字符串');

  }

  function semaphore(n) { var busy = 0, waiters = []; return { acquire: function () { if (busy < n) { busy++; return Promise.resolve(); } return new Promise(function (res) { waiters.push(res); }); }, release: function () { busy--; var w = waiters.shift(); if (w) { busy++; w(); } } }; }

  function chatCompletion(cfg, messages, temperature) {
    return new Promise(function (resolve, reject) {
      /* apiBase 允许裸主机（缺省补 /v1/openai）或已带路径的 OpenAI 兼容 base */
      var ub = new URL(cfg.apiBase);
      var bp = ub.pathname.replace(/\/+$/, '');
      var u = new URL((bp || '/v1/openai') + '/chat/completions', ub);
      var mod = u.protocol === 'http:' ? http : https;
      var body = JSON.stringify({ model: cfg.model, messages: messages, temperature: temperature, max_tokens: 8192, chat_template_kwargs: { enable_thinking: false } });
      var req = mod.request({ hostname: u.hostname, port: u.port || (u.protocol === 'http:' ? 80 : 443), path: u.pathname, method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + cfg.apiKey, 'Content-Length': Buffer.byteLength(body) }, timeout: 300000 }, function (res) {
        var d = ''; res.on('data', function (c) { d += c; });
        res.on('end', function () {
          var responseBody = d;
          try { responseBody = JSON.parse(d); } catch (_) {}
          try { diagnostic('response', { status: res.statusCode, body: responseBody }); } catch (_) { return reject(fatal('诊断文件写入失败')); }
          if (res.statusCode === 429 || res.statusCode >= 500) return reject(new Error('HTTP ' + res.statusCode));
          if (res.statusCode !== 200) return reject(fatal('HTTP ' + res.statusCode));
          try {
            var j = JSON.parse(d), choice = j.choices && j.choices[0];
            if (!choice || choice.finish_reason !== 'stop' || !choice.message || choice.message.refusal || typeof choice.message.content !== 'string' || !choice.message.content.trim()) throw fatal('模型响应未正常完成（finish_reason=' + (choice && choice.finish_reason) + '）');
            if (redact(choice.message.content) !== choice.message.content) throw fatal('模型响应含敏感凭据，已拒绝写出');
            resolve({ content: choice.message.content, usage: j.usage || {}, responseModel: j.model || null });
          } catch (e) { reject(e.fatal ? e : fatal('模型响应不是有效 JSON/完成结果')); }
        });
      });
      req.on('timeout', function () { req.destroy(new Error('request timeout')); });
      req.on('error', reject);
      diagnostic('request', { model: cfg.model, messages: messages, temperature: temperature, max_tokens: 8192 });
      req.write(body); req.end();
    });
  }
  async function callWithRetry(cfg, messages, temperature) {
    var delays = [2000, 4000, 8000, 16000];

  function loadGlossary(rc) {

  async function main() {

    secrets.push(meta.token);
    var entries = parseSrt(fs.readFileSync(P.srtPath, 'utf8'));
    /* diarization 剥离：[SPEAKER_NN]: 前缀 → speaker 字段（最终字幕不显示，仅供翻译上下文）；
     * 可信性判定：<2 人无信息量；切换率 >=30/分钟 = 重叠语音/标注混乱 → 降级剥离（用户约定：重叠语音不强制标说话人） */
    var speakerInfo = (function () {
      var set = {}, last = null, switches = 0, marked = 0;
      entries.forEach(function (e) {
        var m = (e.text || '').match(/^\[SPEAKER_(\d+)\]:?\s*/);
        if (!m) return;
        marked++;
        e.speaker = m[1];
        e.text = e.text.slice(m[0].length).trim();
        if (last !== null && m[1] !== last) switches++;
        last = m[1];
        set[m[1]] = 1;
      });
      var count = Object.keys(set).length;
      var durMin = entries.length ? (entries[entries.length - 1].end - entries[0].start) / 60000 : 0;
      var rate = durMin > 0 ? switches / durMin : 0;
      return { diarized: marked > 0, speakers: count, switches: switches, switchRatePerMin: Math.round(rate * 10) / 10, usable: count >= 2 && rate < 30 };
    })();
    var useSpeakers = speakerInfo.diarized && speakerInfo.usable;
    if (speakerInfo.diarized && !useSpeakers) entries.forEach(function (e) { delete e.speaker; });
    var warnings = timelineWarnings(entries);
    await postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, timelineWarnings: warnings });

    /* 运行配置（术语表/粘贴文本）经 invoke 自取，P 只承载有界字段 */
    var rc = await invokeGet(meta, P.callbackPluginId, 'getRunConfig', {});
    if (!rc || typeof rc !== 'object') throw fatal('运行配置（术语表）读取失败');
    reviewDir = path.join(P.fmbDataDir, 'subtitle-reviews', crypto.randomUUID());
    fs.mkdirSync(reviewDir, { recursive: true });
    report = { version: 1, taskId: P.taskId, startedAt: new Date().toISOString(), status: 'translating', reviewStatus: 'in_progress', timelineWarnings: warnings,
      config: { model: P.model || 'Qwen/Qwen2.5-72B-Instruct', apiBase: apiBase, sourceLang: sourceLang || 'unknown', semanticReview: !!P.semanticReview, retainDiagnostics: !!P.retainDiagnostics, promptVersion: 'json-contract-v3', reviewContractVersion: 'quality-v1', reviewCategories: Object.keys(REVIEW_CRITERIA), revisionPolicy: 'mark-pending-no-extra-review', chunkEntries: 20, chunkChars: 2000, speakerInfo: speakerInfo, concurrency: concurrency },
      references: { pastedSha256: rc.glossary ? crypto.createHash('sha256').update(rc.glossary).digest('hex') : null, files: [] },
      entries: entries.map(function (e) { return { id: e.id, start: e.start, end: e.end, source: e.text, translation: '', speaker: e.speaker || null }; }) };
    var glossary = loadGlossary(rc);
    var initialReviewPath = saveReport();
    await postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, reviewPath: initialReviewPath, reviewStatus: 'in_progress', reviewSummary: '本次翻译/复核尚未完成', sourceLang: sourceLang || 'unknown', model: report.config.model });
    /* 零条目 srt 合法（全静音媒体）：写空 translated.srt 并直接报告成功，不取 key、不调 LLM */
    if (!entries.length) {
      var emptyPath = path.join(P.workDir, 'translated.srt');
      fs.writeFileSync(emptyPath, '', 'utf8');
      report.status = '空字幕（未调用模型）'; report.reviewStatus = 'not_run'; var emptyReviewPath = saveReport();
      await postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, reviewStatus: report.reviewStatus, reviewSummary: report.status });
      console.log(JSON.stringify({ ok: true, translatedSrtPath: emptyPath, lineCount: 0, chunks: 0, tokens: 0 }));
      await postInvoke(meta, P.callbackPluginId, 'storeResult', { taskId: P.taskId, ok: true, translatedSrtPath: emptyPath, lineCount: 0, chunks: 0, usage: { total_tokens: 0 }, reviewPath: emptyReviewPath });
      return;
    }
    var keyRes = await invokeGet(meta, P.callbackPluginId, 'getApiKey', {});
    if (!keyRes || !keyRes.key) throw fatal('未配置 DeepInfra API Key（在 Studio 配置页填写）');
    secrets.push(keyRes.key);
    var cfg = { apiKey: keyRes.key, apiBase: apiBase, model: report.config.model };
    var src = ({ ja: '日语', en: '英语' })[sourceLang] || sourceLang;
    var chunks = chunkEntries(entries, 20, 2000);
    // Splitting only recovers structural failures. Context is immutable even after earlier chunks translate.
    function ctxSlice(from, to) {
      var out = [];
      for (var i = Math.max(0, from); i < to && i < entries.length; i++) {
        var cue = { id: entries[i].id, source: origTexts[i] };
        if (useSpeakers && entries[i].speaker) cue.speaker = 'SPEAKER_' + entries[i].speaker;
        out.push(cue);
      }
      return out;
    }
    async function translateRange(offset, count, ci, depth, tokenBox) {
      /* 上下文/待译条目一律取 origTexts 快照：前序块完成后 entries.text 已是译文（SUCCUBUSQ Cue 不可变语义） */
      /* 二分子块断点兜底：主块 key 未命中（上轮靠二分完成）时子块可整段命中，避免契约重试白跑 */
      if (applyDone(offset, count)) return entries.slice(offset, offset + count).map(function (e) { return e.text; });
      var cues = ctxSlice(offset, offset + count);
      var payload = {
        reference: glossary,
        before_context: ctxSlice(Math.max(0, offset - 3), offset),
        cues: cues,
        after_context: ctxSlice(offset + count, offset + count + 3),
      };
      var messages = [{ role: 'system', content: buildSystem(src, useSpeakers) }, { role: 'user', content: JSON.stringify(payload) }];
      var translated = null, lastErr = null, lastContent = '', allResponses = [];
      for (var attempt = 0; attempt < 3 && !translated; attempt++) {
        try {
          var temp = attempt === 0 ? 0.3 : 0.15;
          var msgs = attempt === 0 ? messages : messages.concat([
            { role: 'assistant', content: lastContent },
            { role: 'user', content: '上次输出未通过结构校验：' + (lastErr && lastErr.message) + '。请按原始 cues 重新输出完整 JSON。' },
          ]);
          var r = await trackedCall('translate', msgs, temp);
          lastContent = r.content;
          allResponses.push(r.content);
          tokenBox.total += (r.usage && r.usage.total_tokens) || 0;
          var map = parseTranslations(r.content, cues);
          translated = cues.map(function (c) { return map[c.id]; });
          if (r.responseModel) report.responseModel = r.responseModel;
        } catch (e) { diagnostic('rejected', { offset: offset, count: count, attempt: attempt, error: e.message }); if (e.fatal) throw e; lastErr = e; }
        await postInvoke(meta, P.callbackPluginId, 'storeProgress', { taskId: P.taskId, ts: Date.now() });
      }
      if (translated) {
        /* 块级断点（含二分子块）：每块完成即落盘，重跑时跳过已完成块（canan FC 实测：135 块全量重翻 3 次烧掉数小时） */
        try {
          var rec = { key: entries[offset].id + '-' + entries[offset + count - 1].id, texts: {} };
          cues.forEach(function (c, i2) { rec.texts[c.id] = translated[i2]; });
          fs.appendFileSync(progressPath, JSON.stringify(rec) + '\n', 'utf8');
        } catch (_) {}
        return translated;
      }
      /* 深度兜底恢复：3 次严格重试均失败时（典型场景：模型对 source 反复规范化无法逐字回显，
       * 即便单条 + 反馈重试也救不回来），尝试放宽解析——已知正确 source（origTexts 快照），
       * 跳过 source 比对，只验证 id 集合一致 + text 非空。复用已有响应（不再调 API）；
       * 恢复成功会写 diagnostics 标记 source-recovered，人工在 review.html 复核该条。 */
      if (count <= 1 || depth >= 6) {
        for (var ri = 0; ri < allResponses.length; ri++) {
          try {
            var lenientMap = parseTranslationsLenient(allResponses[ri], cues);
            var recovered = cues.map(function (c) { return lenientMap[c.id]; });
            diagnostic('source-recovered', { offset: offset, count: count, responseIndex: ri, originalError: lastErr && lastErr.message });
            try {
              var rec2 = { key: entries[offset].id + '-' + entries[offset + count - 1].id, texts: {} };
              cues.forEach(function (c, i3) { rec2.texts[c.id] = recovered[i3]; });
              fs.appendFileSync(progressPath, JSON.stringify(rec2) + '\n', 'utf8');
            } catch (_) {}
            return recovered;
          } catch (_) {}
        }
        throw fatal('块 ' + (ci + 1) + '/' + chunks.length + ' 翻译失败: ' + (lastErr && lastErr.message));
      }
      console.error('split retry: block ' + (ci + 1) + ' lines ' + (offset + 1) + '-' + (offset + count) + ' depth ' + depth + ' (' + (lastErr && lastErr.message) + ')');
      var mid = Math.floor(count / 2);
      var a = await translateRange(offset, mid, ci, depth + 1, tokenBox);
      await postInvoke(meta, P.callbackPluginId, 'storeProgress', { taskId: P.taskId, ts: Date.now() });
      var b = await translateRange(offset + mid, count - mid, ci, depth + 1, tokenBox);
      return a.concat(b);
    }
    /* 断点续跑：raw.srt 复用使 chunking 确定性一致，块 key = (首id,尾id) */
    var progressPath = path.join(P.workDir, 'translate-progress.ndjson');
    var doneRanges = {};
    try {
      fs.readFileSync(progressPath, 'utf8').split(/\r?\n/).forEach(function (line) {
        if (!line.trim()) return;
        try { var rec = JSON.parse(line); if (rec && rec.key && rec.texts) doneRanges[rec.key] = rec.texts; } catch (_) {}
      });
    } catch (_) {}
    function applyDone(offset, count) {
      var key = entries[offset].id + '-' + entries[offset + count - 1].id;
      if (!(key in doneRanges)) return false;
      for (var i = 0; i < count; i++) {
        var txt = doneRanges[key][entries[offset + i].id];
        if (typeof txt !== 'string') return false;
        entries[offset + i].text = txt;
        report.entries[offset + i].translation = txt;
      }
      return true;
    }
    var tokenBox = { total: 0 };
    /*
     * 并行流水线：块间零数据依赖（翻译/复核上下文一律取 origTexts 快照，复核只读本块译文），
     * 翻译池 + 复核池各 concurrency 路并行推进。质量不变：每次调用的输入与串行执行完全一致，
     * 只有执行顺序变化。429/5xx 由 callWithRetry 抖动退避兜底；断点语义不变（块级 key，重跑跳过已完成块）。
     */
    var stats = { translated: 0, reviewed: 0, resumed: 0, total: chunks.length };
    var liveCalls = []; /* {kind, at}：在途调用子状态（进度文本显示存活秒数） */
    function reportProgress() {
      var parts = [P.semanticReview ? '已译 ' + stats.translated + '/' + stats.total + ' · 已复核 ' + stats.reviewed + '/' + stats.total : '已完成 ' + stats.translated + '/' + stats.total];
      var groups = {};
      liveCalls.forEach(function (c) { (groups[c.kind] = groups[c.kind] || []).push(Math.round((Date.now() - c.at) / 1000)); });
      var bits = [];
      [['translate', '翻译'], ['review', '复核'], ['redo', '修订']].forEach(function (kv) {
        if (!groups[kv[0]]) return;
        var ages = groups[kv[0]];
        bits.push(kv[1] + '×' + ages.length + (ages.length === 1 ? '(' + ages[0] + 's)' : '(最长' + Math.max.apply(null, ages) + 's)'));
      });
      if (bits.length) parts.push('在途 ' + bits.join(' '));
      if (stats.resumed) parts.push('断点续跑 ' + stats.resumed + ' 块');
      postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, stage: 'translating', text: '翻译中 · ' + parts.join(' · ') });
      postInvoke(meta, P.callbackPluginId, 'storeProgress', { taskId: P.taskId, ts: Date.now() });
    }
    async function trackedCall(kind, messages, temperature) {
      var rec = { kind: kind, at: Date.now() };
      liveCalls.push(rec);
      reportProgress();
      try { return await callWithRetry(cfg, messages, temperature); }
      finally { var ix = liveCalls.indexOf(rec); if (ix >= 0) liveCalls.splice(ix, 1); reportProgress(); }
    }
    var offsets = chunks.map(function (c) { return entries.indexOf(c[0]); });
    var translateSlots = semaphore(concurrency);
    var reviewQ = [], reviewWaiters = [];
    function enqueueReview(ci) { if (reviewWaiters.length) reviewWaiters.shift()(ci); else reviewQ.push(ci); }
    function takeReview() { return reviewQ.length ? Promise.resolve(reviewQ.shift()) : new Promise(function (res) { reviewWaiters.push(res); }); }
    async function translateBlockTask(ci) {
      await translateSlots.acquire();
      try {
        var chunk = chunks[ci], offset = offsets[ci];
        if (applyDone(offset, chunk.length)) stats.resumed++;
        else {
          var translated = await translateRange(offset, chunk.length, ci, 0, tokenBox);
          for (var k2 = 0; k2 < chunk.length; k2++) { chunk[k2].text = translated[k2]; report.entries[offset + k2].translation = translated[k2]; }
        }
        saveReport();
        stats.translated++;
        reportProgress();
      } finally { translateSlots.release(); }
      enqueueReview(ci);
    }
    async function finishBlock(ci) {
      var chunk = chunks[ci], offset = offsets[ci];
      try {
        if (P.semanticReview) {
          var reviewPayload = { reference: glossary, speaker_info: speakerInfo, before_context: ctxSlice(offset - 3, offset), cues: report.entries.slice(offset, offset + chunk.length), after_context: ctxSlice(offset + chunk.length, offset + chunk.length + 3) };
          var reviewMessages = [{ role: 'system', content: buildReviewSystem() }, { role: 'user', content: JSON.stringify(reviewPayload) }];
          var reviewed = await trackedCall('review', reviewMessages, 0);
          tokenBox.total += reviewed.usage.total_tokens || 0;
          var judgments;
          try { judgments = JSON.parse(reviewed.content); } catch (_) { throw fatal('对齐复核输出不是 JSON'); }
          var reviewIds = new Set(chunk.map(function (e) { return e.id; }));
          if (!Array.isArray(judgments) || judgments.length !== chunk.length) throw fatal('对齐复核编号集合不完整');
          judgments.forEach(function (j) {
            if (!j || !reviewIds.delete(j.id) || typeof j.verdict !== 'string' || !Object.prototype.hasOwnProperty.call(REVIEW_CRITERIA, j.verdict) || typeof j.reason !== 'string' || (j.verdict !== 'ok' && !j.reason.trim())) throw fatal('对齐复核编号/结论/依据无效');
            var row = report.entries.find(function (e) { return e.id === j.id; }); row.verdict = j.verdict; row.reason = j.reason; row.reviewState = j.verdict === 'ok' ? 'no_issues_detected' : 'needs_attention';
          });
          /* Only existing structural flags trigger one rewrite. Other quality findings need human/audio review. */
          var bad = judgments.filter(function (j) { return j.verdict === 'shifted' || j.verdict === 'omitted'; });
          if (bad.length) {
            var badIds = new Set(bad.map(function (j) { return j.id; }));
            var redoCues = [], redoPositions = [];
            for (var i2 = offset; i2 < offset + chunk.length; i2++) if (badIds.has(entries[i2].id)) { redoCues.push(ctxSlice(i2, i2 + 1)[0]); redoPositions.push(i2); }
            var firstRedo = redoPositions[0], lastRedo = redoPositions[redoPositions.length - 1];
            var redoPayload = {
              reference: glossary,
              before_context: ctxSlice(firstRedo - 3, firstRedo),
              cues: redoCues,
              after_context: ctxSlice(lastRedo + 1, lastRedo + 4),
              span_context: ctxSlice(firstRedo, lastRedo + 1),
              review_feedback: bad.map(function (j) { return { id: j.id, problem: j.verdict, reason: j.reason }; }),
            };
            var redoMsgs = [
              { role: 'system', content: buildSystem(src, useSpeakers) + '\nspan_context 保留待修订范围内完整原文，包括未点名的中间条目，仅供理解完整词句；只输出 cues 中点名的 ID，不得把孤立词尾当独立话语。以下是复核员对上一版译文的意见：' + JSON.stringify(bad.map(function (j) { return { id: j.id, problem: j.verdict, reason: j.reason }; })) },
              { role: 'user', content: JSON.stringify(redoPayload) },
            ];
            var redo = await trackedCall('redo', redoMsgs, 0.15);
            tokenBox.total += (redo.usage && redo.usage.total_tokens) || 0;
            var redoMap = parseTranslations(redo.content, redoCues);
            bad.forEach(function (j) {
              var i5 = entries.findIndex(function (e) { return e.id === j.id; });
              entries[i5].text = redoMap[j.id];
              var row2 = report.entries.find(function (e) { return e.id === j.id; });
              row2.reviewHistory = [{ verdict: j.verdict, reason: j.reason, translation: row2.translation }];
              row2.translation = redoMap[j.id];
              row2.verdict = 'auto-redone';
              row2.reviewState = 'revised_pending_review';
              row2.reason = '复核意见: ' + j.reason + '（已修订，待复核；未再次调用模型复核）';
            });
          }
        }
        saveReport();
      } finally {
        /* 终检点（含修订后文本；复核失败也保留已译文本，重跑跳过翻译只重复核） */
        var finalCheckpoint = { key: entries[offset].id + '-' + entries[offset + chunk.length - 1].id, texts: {} };
        chunk.forEach(function (e) { finalCheckpoint.texts[e.id] = e.text; });
        fs.appendFileSync(progressPath, JSON.stringify(finalCheckpoint) + '\n', 'utf8');
        stats.reviewed++;
        reportProgress();
      }
    }
    hbTimer = setInterval(reportProgress, 30000); /* 块内活性心跳：长调用期间进度文本也在走秒 */
    try {
      /* 复核池必须先于翻译任务启动：译完一块即入队复核（真流水线）。
       * 若等全部译完再起复核池，翻译与复核永不重叠，总时长≈两阶段之和（实测陷阱）。 */
      var reviewWorkers = [];
      if (P.semanticReview) {
        for (var w = 0; w < concurrency; w++) reviewWorkers.push((async function () {
          for (;;) { var rci = await takeReview(); if (rci < 0) return; await finishBlock(rci); }
        })());
      }
      await Promise.all(chunks.map(function (c, ci) { return translateBlockTask(ci); }));
      if (P.semanticReview) {
        /* 哨兵排在全部真块之后（FIFO），复核池按各块译毕顺序消化 */
        for (var w2 = 0; w2 < concurrency; w2++) enqueueReview(-1);
        await Promise.all(reviewWorkers);
      } else {
        for (var ci2 = 0; ci2 < chunks.length; ci2++) await finishBlock(ci2);
      }
    } finally { clearInterval(hbTimer); }
    var totalTokens = tokenBox.total;
    var outPath = path.join(P.workDir, 'translated.srt');
    fs.writeFileSync(outPath, formatSrt(entries), 'utf8');
    var pending = report.entries.filter(function (e) { return e.verdict === 'auto-redone'; }).length;
    var attention = report.entries.filter(function (e) { return e.verdict && e.verdict !== 'ok' && e.verdict !== 'auto-redone'; }).length;
    report.reviewStatus = !P.semanticReview ? 'not_run' : pending ? 'pending_recheck' : attention ? 'needs_attention' : 'no_issues_detected';
    report.status = !P.semanticReview ? '结构校验通过，语义待人工复核' : pending ? '已修订，待复核（' + pending + ' 条修订未复核；另有 ' + attention + ' 条待确认）' : attention ? '有 ' + attention + ' 条质量疑点，待人工或原音核查' : '模型未检出所列问题，仍需人工结合原音确认';
    report.totalTokens = totalTokens;
    var reviewPath = saveReport();
    console.log(JSON.stringify({ ok: true, translatedSrtPath: outPath, lineCount: entries.length, chunks: chunks.length, tokens: totalTokens }));
    await postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, reviewStatus: report.reviewStatus, reviewSummary: report.status });
    await postInvoke(meta, P.callbackPluginId, 'storeResult', { taskId: P.taskId, ok: true, translatedSrtPath: outPath, lineCount: entries.length, chunks: chunks.length, usage: { total_tokens: totalTokens }, reviewPath: reviewPath });
  }
  main().then(function () { process.exit(0); }).catch(async function (e) {
    var error = redact(String(e && e.message || e)), reviewPath = '';
    try { if (report) { report.status = 'failed'; report.reviewStatus = 'failed'; report.error = error; reviewPath = saveReport(); diagnostic('failure', { error: error }); } } catch (_) {}
    console.error('llm failed: ' + error);
    try { var meta = findHttpMeta(P.fmbDataDir); if (meta) { await postInvoke(meta, P.studioPluginId, 'storeProgress', { taskId: P.taskId, stage: 'error', text: error, reviewPath: reviewPath, reviewStatus: 'failed', reviewSummary: '本次翻译/复核失败，请查看错误及报告' }); await postInvoke(meta, P.callbackPluginId, 'storeResult', { taskId: P.taskId, ok: false, error: error, reviewPath: reviewPath }); } } catch (_) {}
    setTimeout(function () { process.exit(1); }, 1000);
  });
})();
